cmake_minimum_required(VERSION 3.22)
project(cinemaair C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(V ${CMAKE_CURRENT_SOURCE_DIR}/../vendor)
file(GLOB PLIST ${V}/libplist/src/*.c)
list(APPEND PLIST ${V}/libplist/libcnary/node.c ${V}/libplist/libcnary/node_list.c)
add_library(plist STATIC ${PLIST})
target_include_directories(plist PUBLIC ${V}/libplist/include PRIVATE ${V}/libplist/libcnary/include ${V}/libplist/src)
target_compile_definitions(plist PRIVATE PACKAGE_VERSION="2.7.0" HAVE_STRNDUP HAVE_MEMMEM HAVE_STRPTIME HAVE_TIMEGM HAVE_LOCALTIME_R HAVE_GMTIME_R HAVE_TM_TM_GMTOFF HAVE_TM_TM_ZONE _GNU_SOURCE)
file(GLOB CORE ${V}/uxplay/lib/*.c ${V}/uxplay/lib/playfair/*.c ${V}/uxplay/lib/llhttp/*.c ${V}/uxplay/lib/mdnsd/*.c)
add_library(airplay STATIC ${CORE})
target_include_directories(airplay PUBLIC ${V}/uxplay/lib PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${V}/uxplay/lib/playfair ${V}/uxplay/lib/llhttp)
target_compile_definitions(airplay PRIVATE PLIST_210 PLIST_230 OPENSSL_API_COMPAT=0x10101000L)
if(ANDROID)
  add_library(crypto STATIC IMPORTED)
  set_target_properties(crypto PROPERTIES IMPORTED_LOCATION ${CMAKE_CURRENT_SOURCE_DIR}/../.local/openssl/${ANDROID_ABI}/lib/libcrypto.a INTERFACE_INCLUDE_DIRECTORIES ${CMAKE_CURRENT_SOURCE_DIR}/../.local/openssl/${ANDROID_ABI}/include)
else()
  find_package(OpenSSL 3 REQUIRED)
  add_library(crypto ALIAS OpenSSL::Crypto)
endif()
target_link_libraries(airplay PUBLIC plist crypto m)
if(ANDROID)
  set(AUDIO_ABI ${ANDROID_ABI})
else()
  if(NOT AUDIO_HOST_ABI)
    set(AUDIO_HOST_ABI host)
  endif()
  set(AUDIO_ABI ${AUDIO_HOST_ABI})
endif()
set(AUDIO_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../.local/audio/${AUDIO_ABI})
add_library(avcodec STATIC IMPORTED)
set_target_properties(avcodec PROPERTIES IMPORTED_LOCATION ${AUDIO_ROOT}/lib/libavcodec.a INTERFACE_INCLUDE_DIRECTORIES ${AUDIO_ROOT}/include)
add_library(avutil STATIC IMPORTED)
set_target_properties(avutil PROPERTIES IMPORTED_LOCATION ${AUDIO_ROOT}/lib/libavutil.a)
add_library(alac STATIC alac_decoder.cpp)
target_link_libraries(alac PUBLIC avcodec avutil m)
if(ANDROID)
  add_library(cinemaair SHARED bridge.cpp)
  target_link_libraries(cinemaair airplay alac log)
else()
  add_executable(protocol-test ../tests/protocol_test.c)
  target_link_libraries(protocol-test airplay pthread)
  add_executable(server-test ../tests/server_test.c)
  target_link_libraries(server-test airplay pthread)
  target_include_directories(server-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  add_executable(alac-test ../tests/alac_test.cpp)
  target_link_libraries(alac-test alac pthread)
  target_include_directories(alac-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  enable_testing()
  add_test(NAME bounded-http COMMAND protocol-test)
  add_test(NAME rtsp-start-stop COMMAND server-test)
  add_test(NAME alac-roundtrip COMMAND alac-test ${CMAKE_CURRENT_SOURCE_DIR}/../tests/fixtures/alac.packet ${CMAKE_CURRENT_SOURCE_DIR}/../tests/fixtures/alac.pcm)
endif()
